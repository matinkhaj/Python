a = 10
b = 5

print(a + b)
# جمع دو متغیر a و b

print(a - b)
# تفریق متغیر b از متغیر a

print(a * b)
# ضرب دو متغیر در یکدیگر

print(a / b)
# تقسیم متغیر a بر متغیر b

print(a ** b)
# محاسبه توان؛ یعنی a به توان b

a = 10
b = 5

print(a > b)
# بررسی می‌کند آیا a بزرگ‌تر از b است یا خیر

print(a < b)
# بررسی می‌کند آیا a کوچک‌تر از b است یا خیر

print(a == b)
# بررسی می‌کند آیا مقدار دو متغیر با هم برابر است یا خیر

print(a != b)
# بررسی می‌کند آیا مقدار دو متغیر با هم متفاوت است یا خیر

print(a >= b)
# بررسی می‌کند آیا a بزرگ‌تر یا مساوی b است یا خیر

x = 4
y = 2

print(2 * x + 3 * y)
# محاسبه عبارت جبری 2x + 3y

print(x ** 2 + y ** 2)
# محاسبه مجموع مربع دو متغیر x و y

print(3 * x - 2 * y)
# محاسبه عبارت جبری 3x - 2y

print((x + y) * 2)
# ابتدا مجموع x و y محاسبه شده و سپس در 2 ضرب می‌شود

print((x ** 2 - y ** 2) / (x + y))
# محاسبه تفاضل مربع دو متغیر و تقسیم آن بر مجموع آن‌ها
